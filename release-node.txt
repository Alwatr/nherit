# Remember: change logs generated from pull requests, do not push hotfix directly to default branch.

current_version=4.13.0
new_version=4.13.1

git checkout next
git pull

find . -name 'Dockerfile' -exec sed -i '' "s/$current_version/$new_version/g" {} +

git commit -am "release: v$new_version"
git push
git tag v$new_version -m v$new_version --force --sign
git push origin v$new_version --force

gh release create v$new_version --generate-notes
gh workflow view 'Build & Publish Containers'
gh release view -w
